cmake_minimum_required(VERSION 3.16)
project(bof_vision)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(bof_interfaces REQUIRED)
add_subdirectory(${CMAKE_CURRENT_SOURCE_DIR}/core ${CMAKE_CURRENT_BINARY_DIR}/bof_core)
find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio imgcodecs dnn)
# An optional executable that is switched off must not survive from an earlier build: bof_core is
# static, so a stale binary carries old code (symlink-install would keep pointing at it).
function(bof_remove_stale_executable name)
  file(REMOVE "${CMAKE_CURRENT_BINARY_DIR}/${name}")
  install(CODE "file(REMOVE \"\${CMAKE_INSTALL_PREFIX}/lib/${PROJECT_NAME}/${name}\")")
endfunction()
foreach(node source_node pipeline_node aim_node)
  add_executable(${node} src/${node}.cpp)
  ament_target_dependencies(${node} rclcpp rcl_interfaces sensor_msgs std_msgs diagnostic_msgs cv_bridge bof_interfaces)
  target_link_libraries(${node} bof_core)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
if(BOF_WITH_MVS)
  add_executable(hik_camera_node src/hik_camera_node.cpp)
  ament_target_dependencies(hik_camera_node rclcpp rcl_interfaces sensor_msgs std_msgs diagnostic_msgs cv_bridge)
  target_link_libraries(hik_camera_node bof_core)
  install(TARGETS hik_camera_node DESTINATION lib/${PROJECT_NAME})
else()
  bof_remove_stale_executable(hik_camera_node)
endif()
# The field tools link the same core as the ROS nodes.
# Installing under lib/<package> also makes the non-ROS tools available via ros2 run.
foreach(app replay calibrate)
  add_executable(bof_${app} apps/${app}.cpp)
  target_link_libraries(bof_${app} PRIVATE bof_core)
  install(TARGETS bof_${app} DESTINATION lib/${PROJECT_NAME})
endforeach()
# Offline rosbag tools: aim replay/latency analysis and hand-eye calibration.
find_package(rosbag2_cpp REQUIRED)
foreach(app aim_replay handeye)
  add_executable(bof_${app} apps/${app}.cpp)
  ament_target_dependencies(bof_${app} rclcpp rosbag2_cpp sensor_msgs cv_bridge bof_interfaces)
  target_link_libraries(bof_${app} bof_core)
  install(TARGETS bof_${app} DESTINATION lib/${PROJECT_NAME})
endforeach()
option(BOF_BUILD_LIVE "Build the camera/video field-debug application" ON)
if(BOF_BUILD_LIVE)
  find_package(OpenCV REQUIRED COMPONENTS highgui)
  add_executable(bof_live apps/live.cpp)
  target_link_libraries(bof_live PRIVATE bof_core opencv_highgui)
  install(TARGETS bof_live DESTINATION lib/${PROJECT_NAME})
else()
  bof_remove_stale_executable(bof_live)
endif()
install(TARGETS bof_core EXPORT bof_core_targets DESTINATION lib)
install(DIRECTORY core/include/ DESTINATION include)

ament_export_targets(bof_core_targets HAS_LIBRARY_TARGET)
ament_export_dependencies(OpenCV Eigen3 Threads yaml-cpp)
if(BOF_WITH_OPENVINO)
  ament_export_dependencies(OpenVINO)
endif()
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(core_test test/unit/core_test.cpp test/unit/contracts_test.cpp test/unit/config_test.cpp test/unit/aim_test.cpp)
  target_link_libraries(core_test bof_core)
endif()
ament_package()
