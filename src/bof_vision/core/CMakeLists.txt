find_package(OpenCV REQUIRED COMPONENTS core imgproc video videoio imgcodecs dnn calib3d)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
find_package(yaml-cpp REQUIRED)
option(BOF_WITH_OPENVINO "Enable Tongji YOLOv5 OpenVINO adapter" OFF)
add_library(bof_core src/aim/config.cpp src/aim/tracker.cpp src/aim/fire.cpp src/aim/engine.cpp src/aim/overlay.cpp src/io/gimbal_link.cpp src/core.cpp src/tracking/pixel_tracker.cpp src/config.cpp src/geometry.cpp src/pose.cpp src/pipeline/pipeline.cpp src/pipeline/pipeline_switcher.cpp src/detection/yolov5_decode.cpp src/detection/keypoint_decode.cpp src/detection/detector_factory.cpp src/io/camera_factory.cpp src/io/recorder.cpp)
target_compile_features(bof_core PUBLIC cxx_std_17)
target_compile_options(bof_core PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(bof_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> ${OpenCV_INCLUDE_DIRS})
target_link_libraries(bof_core PUBLIC ${OpenCV_LIBS} Eigen3::Eigen Threads::Threads yaml-cpp)
if(BOF_WITH_OPENVINO)
  find_package(OpenVINO REQUIRED COMPONENTS Runtime)
  target_sources(bof_core PRIVATE src/detection/tongji_detector.cpp)
  target_link_libraries(bof_core PUBLIC openvino::runtime)
  target_compile_definitions(bof_core PRIVATE BOF_WITH_OPENVINO)
endif()
option(BOF_WITH_MVS "Enable optional Hikrobot capture adapter" OFF)
set(MVS_ROOT "/opt/MVS" CACHE PATH "Installed MVS SDK root")
if(BOF_WITH_MVS)
  find_path(MVS_INCLUDE_DIR MvCameraControl.h HINTS "${MVS_ROOT}/include" REQUIRED)
  find_library(MVS_LIBRARY MvCameraControl HINTS "${MVS_ROOT}/lib/64"
    "${MVS_ROOT}/lib/amd64" "${MVS_ROOT}/lib/aarch64" "${MVS_ROOT}/lib/arm64" REQUIRED)
  target_sources(bof_core PRIVATE src/io/hik_source.cpp)
  target_compile_definitions(bof_core PRIVATE BOF_WITH_MVS)
  target_include_directories(bof_core PRIVATE ${MVS_INCLUDE_DIR})
  target_link_libraries(bof_core PUBLIC ${MVS_LIBRARY})
endif()
